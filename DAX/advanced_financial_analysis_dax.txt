-- ============================================================
-- 1. Total Revenue
-- ============================================================

Total Revenue =
SUM(FinancialData[Revenue])

-- ============================================================
-- 2. Total Profit
-- ============================================================

Total Profit =
SUM(FinancialData[Profit])

-- ============================================================
-- 3. Profit Margin
-- ============================================================

Profit Margin % =
DIVIDE(
    [Total Profit],
    [Total Revenue],
    0
)

-- ============================================================
-- 1. Total Revenue
-- ============================================================

Total Revenue =
SUM(FinancialData[Revenue])

-- ============================================================
-- 2. Total Profit
-- ============================================================

Total Profit =
SUM(FinancialData[Profit])

-- ============================================================
-- 3. Profit Margin
-- ============================================================

Profit Margin % =
DIVIDE(
    [Total Profit],
    [Total Revenue],
    0
)

-- ============================================================
-- 4. Revenue Contribution %
-- ============================================================

Revenue Contribution % =
DIVIDE(
    [Total Revenue],
    CALCULATE(
        [Total Revenue],
        REMOVEFILTERS(FinancialData)
    ),
    0
)

-- ============================================================
-- 5. Profit Contribution %
-- ============================================================

Profit Contribution % =
DIVIDE(
    [Total Profit],
    CALCULATE(
        [Total Profit],
        REMOVEFILTERS(FinancialData)
    ),
    0
)

-- ============================================================
-- 6. Revenue vs Selected Total
-- ============================================================

Revenue % Selected =
DIVIDE(
    [Total Revenue],
    CALCULATE(
        [Total Revenue],
        ALLSELECTED(FinancialData)
    ),
    0
)
